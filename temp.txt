Replace `MagicArtifact` with a real Three.js Saturn: a pearl sphere, thin blush-and-silver rings, gentle floating, and drag-to-orbit interaction. Pause and reset controls keep it usable on touch devices too. Your existing CTA layout stays unchanged.

Install the 3D dependencies:

```bash
npm install three @react-three/fiber @react-three/drei
npm install -D @types/three
```

Replace the React and Framer Motion imports, and add these imports. Keep your existing `ExternalLink` and `site` imports.

```tsx
import { useRef, useState, type ComponentRef } from 'react';
import { motion, useReducedMotion } from 'framer-motion';
import { Canvas } from '@react-three/fiber';
import { Float, OrbitControls } from '@react-three/drei';
import { DoubleSide } from 'three';
import { Pause, Play, RotateCcw } from 'lucide-react';
```

Replace only the existing `MagicArtifact` function:

```tsx
function MagicArtifact() {
  const reducedMotion = useReducedMotion();
  const [paused, setPaused] = useState(false);
  const controls = useRef<ComponentRef<typeof OrbitControls>>(null);
  const moving = !reducedMotion && !paused;
  const buttonClass = 'flex h-9 w-9 items-center justify-center rounded-full border border-white/30 bg-white/10 text-white transition-colors hover:bg-white/25 focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-4 focus-visible:outline-white';

  return (
    <div className="relative mx-auto h-[310px] w-full max-w-[400px] sm:h-[370px]">
      <Canvas
        shadows
        dpr={[1, 1.75]}
        camera={{ position: [0, 1.6, 8], fov: 38 }}
        gl={{ alpha: true, antialias: true }}
        className="cursor-grab active:cursor-grabbing"
        role="img"
        aria-label="Interactive pearl Saturn with blush and silver rings"
        fallback={<div className="p-8 text-center text-white/80">3D preview unavailable</div>}
      >
        <ambientLight intensity={0.85} />
        <directionalLight position={[3, 5, 4]} intensity={3} castShadow
          shadow-mapSize={[1024, 1024]} shadow-bias={-0.0002} />
        <directionalLight position={[-4, 1, -2]} intensity={2} color="#a8e6da" />
        <Float speed={moving ? 1.3 : 0} rotationIntensity={0.12} floatIntensity={0.3}>
          <group rotation={[0.25, 0, -0.35]}>
            <mesh castShadow receiveShadow scale={[1, 0.92, 1]}>
              <sphereGeometry args={[1, 64, 48]} />
              <meshPhysicalMaterial color="#f5ecef" roughness={0.38}
                metalness={0.15} clearcoat={0.65} clearcoatRoughness={0.25} />
            </mesh>
            {([
              [1.3, 1.47, '#dce9e6'],
              [1.5, 1.78, '#e8bacb'],
              [1.81, 1.96, '#f2e8ed'],
              [2.01, 2.06, '#b6d8d1'],
            ] as const).map(([inner, outer, color]) => (
              <mesh key={inner} rotation={[-Math.PI / 2, 0, 0]} castShadow receiveShadow>
                <ringGeometry args={[inner, outer, 128]} />
                <meshStandardMaterial color={color} side={DoubleSide}
                  roughness={0.42} metalness={0.25} />
              </mesh>
            ))}
          </group>
        </Float>
        <OrbitControls ref={controls} enablePan={false} enableZoom={false}
          enableDamping dampingFactor={0.06} autoRotate={moving} autoRotateSpeed={0.65} />
      </Canvas>
      <div className="absolute bottom-0 left-1/2 flex -translate-x-1/2 gap-2">
        <button type="button" className={buttonClass} disabled={!!reducedMotion}
          title={moving ? 'Pause rotation' : 'Resume rotation'}
          aria-label={moving ? 'Pause rotation' : 'Resume rotation'}
          onClick={() => setPaused(value => !value)}>
          {moving ? <Pause size={15} /> : <Play size={15} />}
        </button>
        <button type="button" className={buttonClass} title="Reset view"
          aria-label="Reset view" onClick={() => controls.current?.reset()}>
          <RotateCcw size={15} />
        </button>
      </div>
    </div>
  );
}
```

Automatic motion respects reduced-motion preferences. Scroll zoom is disabled so the object doesn't interfere with page scrolling. I couldn't run or visually verify this against your project here.
